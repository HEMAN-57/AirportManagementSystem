⚡ Event-Driven Database Triggers (PL/pgSQL)
trg_baggage_excess_calc BEFORE INSERT OR UPDATE ON baggage

Executes function fn_calculate_baggage_excess_fee(). When luggage weight exceeds the 15.0 kg allowance, the trigger automatically calculates the excess tariff surcharge at ₹500.00 / kg. 

CREATE OR REPLACE FUNCTION fn_calculate_baggage_excess_fee()
RETURNS TRIGGER AS $$
DECLARE
    excess_kg DECIMAL(5,2);
    tariff_rate DECIMAL(8,2) := 500.00;
BEGIN
    IF NEW.weight_kg > NEW.allowed_weight_kg THEN
        excess_kg := NEW.weight_kg - NEW.allowed_weight_kg;
        NEW.excess_fee_paid := excess_kg * tariff_rate;
    ELSE
        NEW.excess_fee_paid := 0.00;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
